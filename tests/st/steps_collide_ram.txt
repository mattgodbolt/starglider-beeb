# Ramming, on the keys: an object destroyed and scored, an indestructible
# tower at levels 0, 1, 2, 3 and 15 (each level's damage table), Starglider
# One outside the missile view, a walker and a stomper (the shields'
# -8000 each), then the tower again until the ram's damage borrows and the
# game ends.
# The pokes (at a frame's start: tools/st_steps.py) write the player
# ($12E: x, y, z; $13A: the angles), the shields ($260) and the level
# ($296); an object's record is at &5D19A + 64 x its index. The capture
# runs the game slowly, so the waits stand for frames.
wait 38
key 57
wait 3
debug parse {arm}
wait 2
debug echo \nSEG ram3\n
# record 19, a type 3 (x &6E30, z &A40), 34 high: above it at 48, then
# rammed at the ground
poke w w $5c02e $6e30 $ffd0 $0a40
wait 2
poke w w $5c02e $6e30 $ffec $0a40
wait 3
debug echo \nSEG tower\n
# tower 11 (x &1EC0, z &1EC0), at each level in turn, the shields full
# at each
poke w w $5c160 $ffff
poke w w $5c02e $1ec0 $ffd0 $1ec0
poke w w $5c196 0
wait 2.5
poke w w $5c160 $ffff
poke w w $5c196 2
wait 2.5
poke w w $5c160 $ffff
poke w w $5c196 3
wait 2.5
poke w w $5c160 $ffff
poke w w $5c196 15
wait 2.5
poke w w $5c196 1
debug echo \nSEG sg1\n
# Starglider One (record 106) held at x &4000, y -750, z &4000
poke w w $5c160 $ffff
poke w w $5ec2a 0
poke w w $5ec1e $4000 $fd12 $4000
poke w w $5c02e $4000 $fd12 $3fc0
wait 3
debug echo \nSEG walker\n
# the walker (record 49) held at x &5000, z &5000: -8000 a frame and the
# ram's 3000
poke w w $5c160 $ffff
poke w w $5ddea 0
poke w w $5ddde $5000 0 $5000
poke w w $5c02e $5000 $ffd0 $4fc0
wait 3
debug echo \nSEG stomper\n
# the stomper (record 10) held at x &6000, z &6000: a walker too
poke w w $5c160 $ffff
poke w w $5d42a 0
poke w w $5d41e $6000 0 $6000
poke w w $5c02e $6000 $ffd0 $5fc0
wait 3
debug echo \nSEG borrow\n
# the tower at level 2 with less than its 5000 left: the game ends
poke w w $5c02e $1ec0 $ffd0 $1ec0
poke w w $5c196 2
poke w w $5c160 $0fa0
wait 5
debug echo \nSEG end\n
quit
