# The energy-tower group in view, 3,000 units off (x &1EC0, z &1308,
# heading 0), then a tower close up with a tank and a fighter (z &1C70).
wait 38
key 57
wait 4
debug w w $5c02e $1ec0 $ffd0 $1308
debug w w $5c03c 0
wait 0.3
debug parse {arm}
debug echo \nSEG towers\n
wait 6
debug w w $5c02e $1ec0 $ff00 $1c70
debug w w $5c03c 0
debug echo \nSEG closeup\n
wait 4
debug echo \nSEG end\n
quit
