# The new game's set-up, twice, and the level change through every
# layout. Armed on the title: the first game starts from a turn and a
# player position left over (what a mouse on the title, or a game before,
# would leave; poked here, as the keys can't set them there). The second
# starts from what the first left: it ends with Esc (scancode 1, put in
# the last-key latch: the script's keys can't name it) while banking
# left. Then the score's level byte (&0150D) is poked, so world_render
# raises the level: 2, 3, 4, then from 33 to 34 and past it, to the cap.
wait 38
debug w w $12200 $ff70
debug w w $5c02e $1234 $0064 $4321
debug parse {arm}
debug echo \nSEG newgame\n
key 57
wait 3
down 0x4b
wait 3
debug w b $12210 1
debug echo \nSEG title\n
wait 2
up 0x4b
wait 2
key 57
debug echo \nSEG again\n
wait 3
debug w b $11ecb 1
debug echo \nSEG level2\n
wait 3
debug w b $11ecb 2
debug echo \nSEG level3\n
wait 3
debug w b $11ecb 3
debug echo \nSEG level4\n
wait 3
debug w w $5c196 33
debug w b $11ecb 4
debug echo \nSEG level34\n
wait 3
debug w b $11ecb 5
debug echo \nSEG capped\n
wait 3
debug echo \nSEG end\n
quit
